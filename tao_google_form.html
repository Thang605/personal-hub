Trình Tạo Google Form Nhân Sự

Tự động hóa 100% hồ sơ năng lực nhân sự theo mẫu

Mở Google Apps Script (script.new)
⚡ TẠO TỰ ĐỘNG CHỈ VỚI 3 BƯỚC (30 GIÂY)

Khởi tạo biểu mẫu Google Form cho tài khoản Google của bạn

Google Form cần được tạo trực tiếp trên Google Drive của bạn để toàn bộ dữ liệu phản hồi được bảo mật và lưu vào trang tính Google Sheets của bạn. Đoạn mã dưới đây sẽ tự động tạo trọn vẹn biểu mẫu chỉ sau 1 lần bấm Run.

1

Sao chép mã tạo Form

Bấm nút "Copy Mã Script" bên dưới để sao chép toàn bộ mã Apps Script vào bộ nhớ tạm.

2

Mở script.new & Dán mã

Bấm nút mở script.new, dán mã vào trình soạn thảo rồi bấm Ctrl + S để lưu.

3

Bấm "Run" (Chạy)

Bấm Run và cấp quyền. Google Form hoàn chỉnh sẽ xuất hiện ngay trong Google Drive của bạn!

Mở Trang Tạo Script (script.new)
create_personnel_google_form.gs 
function createMergedPersonnelProfileForm() {
  const formTitle = 'PHIẾU THU THẬP & CẬP NHẬT HỒ SƠ NĂNG LỰC CÁ NHÂN';
  const formDescription = 'Biểu mẫu thu thập, cập nhật thông tin cá nhân, trình độ đào tạo, chứng chỉ hành nghề, kỹ năng chuyên môn và nhu cầu đào tạo phục vụ công tác quản lý nhân sự & hồ sơ năng lực công ty.\n\n* Vui lòng điền đầy đủ thông tin vào các mục bên dưới.';
  
  const form = FormApp.create(formTitle);
  form.setDescription(formDescription);
  form.setAllowResponseEdits(true);
  form.setCollectEmail(true);

  // ==========================================
  // PHẦN 1: THÔNG TIN QUẢN TRỊ HỒ SƠ (TIÊU ĐỀ MỤC)
  // ==========================================
  const h1 = form.addSectionHeaderItem();
  h1.setTitle('📋 PHẦN 1: THÔNG TIN QUẢN TRỊ HỒ SƠ')
    .setHelpText('Thông tin mã số và quản lý hồ sơ theo dõi');

  form.addTextItem().setTitle('Mã số hồ sơ').setHelpText('Ví dụ: T27-01').setRequired(true);
  form.addTextItem().setTitle('Số lần cập nhật (Số nhập)').setHelpText('Ví dụ: 1, 2, 5...').setRequired(false);
  form.addDateItem().setTitle('Ngày cập nhật thông tin').setRequired(true);
  form.addScaleItem().setTitle('Mức độ đánh giá / Xếp hạng hiện tại').setBounds(1, 10).setLabels('Cần cải thiện', 'Xuất sắc (10★)').setRequired(false);

  // ==========================================
  // PHẦN 2: MỤC A - THÔNG TIN CÁ NHÂN (TIÊU ĐỀ MỤC)
  // ==========================================
  const h2 = form.addSectionHeaderItem();
  h2.setTitle('👤 MỤC A: THÔNG TIN CÁ NHÂN & QUÁ TRÌNH CÔNG TÁC')
    .setHelpText('Kê khai thông tin cá nhân, văn bằng đào tạo và chứng chỉ');

  form.addTextItem().setTitle('Họ và tên').setHelpText('Ví dụ: ĐỖ VĂN PHONG (Viết hoa đầy đủ)').setRequired(true);
  form.addDateItem().setTitle('Ngày sinh').setHelpText('Định dạng: Ngày/Tháng/Năm (Ví dụ: 10/05/1978)').setRequired(true);
  form.addTextItem().setTitle('Số Căn cước công dân (CCCD / CMND)').setHelpText('Nhập đúng 12 chữ số CCCD (Ví dụ: 052078000567)').setRequired(true);
  form.addTextItem().setTitle('Địa chỉ Email cá nhân / công việc').setHelpText('Ví dụ: Phongt27nt@gmail.com').setRequired(true);
  form.addTextItem().setTitle('Số điện thoại liên hệ & Zalo').setHelpText('Ví dụ: 0913431515; Zalo: 0913431515; 0988112277').setRequired(true);
  form.addParagraphTextItem().setTitle('Chỗ ở hiện tại / Địa chỉ liên hệ').setHelpText('Ví dụ: 38A Lý Tự Trọng, phường Nha Trang, tỉnh Khánh Hòa').setRequired(true);
  form.addTextItem().setTitle('Chức vụ hiện tại').setHelpText('Ví dụ: Chủ tịch HĐQT / Tổng giám đốc / Trưởng phòng / Kỹ sư...').setRequired(true);
  form.addTextItem().setTitle('Số năm công tác tại công ty').setHelpText('Ví dụ: 19 năm 10 tháng').setRequired(true);
  form.addParagraphTextItem().setTitle('Trình độ được đào tạo (Văn bằng, Trường, Chuyên ngành, Năm tốt nghiệp)').setHelpText('Ghi rõ từng văn bằng, số hiệu bằng, ngày cấp.\nVí dụ:\n- Kỹ Sư Giao thông cấp bằng số B139107, ngày 16/10/2000\n- Kỹ Sư đường sắt tốc độ cao, thạc sỹ kỹ thuật cấp bằng số DND.7.7.0013061, ngày 12/11/2018').setRequired(true);
  form.addParagraphTextItem().setTitle('Chứng chỉ hành nghề hoạt động xây dựng / chuyên ngành').setHelpText('Ghi rõ lĩnh vực, hạng cấp, số chứng chỉ, ngày cấp và ngày hết hạn.\nVí dụ:\n- Thiết kế công trình cầu, hầm, đường bộ: Cấp I\n- Tư vấn giám sát công trình giao thông: Cấp I\n- Kỹ Sư định giá: Cấp I\n- Số chứng chỉ: BXD-00020412 | Ngày hết hạn: 05/05/2028').setRequired(false);
  form.addParagraphTextItem().setTitle('Bằng khen, Khen thưởng & Thành tích đạt được').setHelpText('Ghi rõ các bằng khen, cấp khen thưởng và năm nhận.\nVí dụ: Liên hiệp hội năm 2020, Hội KHKT Cầu đường Việt Nam năm 2020; UBND tỉnh khen năm 2021...').setRequired(false);

  // ==========================================
  // PHẦN 3: MỤC B, C & D - KỸ NĂNG & ĐÀO TẠO (TIÊU ĐỀ MỤC)
  // ==========================================
  const h3 = form.addSectionHeaderItem();
  h3.setTitle('💡 MỤC B, C & D: NĂNG LỰC CHUYÊN MÔN & NHU CẦU ĐÀO TẠO')
    .setHelpText('Kỹ năng làm chủ, phần mềm sử dụng và mong muốn bồi dưỡng');

  form.addCheckboxItem().setTitle('B. Khả năng chuyên môn sử dụng THUẦN THỤC').setChoiceValues([
    'Chủ nhiệm Dự án (giao thông: ĐƯỜNG, CẦU, HẦM)',
    'Sử dụng AI trong viết báo cáo (ChatGPT, Gemini, Claude...)',
    'Tin học văn phòng nâng cao (Word, Excel, PowerPoint, Canva)',
    'Quản lý điều hành dự án & Hồ sơ pháp lý xây dựng',
    'Thẩm định thiết kế & Dự toán công trình'
  ]).showOtherOption(true).setRequired(false);

  form.addCheckboxItem().setTitle('C. Khả năng BIẾT CƠ BẢN (Phần mềm & Kỹ thuật)').setChoiceValues([
    'Autodesk Revit',
    'Midas Civil / Midas GTS NX',
    'AutoCAD 2D/3D',
    'Civil 3D',
    'SAP2000 / ETABS'
  ]).showOtherOption(true).setRequired(false);

  form.addCheckboxItem().setTitle('D. Nhu cầu CẦN ĐÀO TẠO / NÂNG CAO KỸ NĂNG').setChoiceValues([
    'Lập trình tự động hóa Python & Phân tích dữ liệu',
    'Ứng dụng AI nâng cao (Claude AI, Prompt Engineering, Tự động hóa quy trình)',
    'BIM nâng cao (Revit Structure, BIM Coordinator)',
    'Midas nâng cao & Tính toán kết cấu phức tạp',
    'Kỹ năng quản trị & Đấu thầu quốc tế FIDIC'
  ]).showOtherOption(true).setRequired(false);

  // ==========================================
  // PHẦN 4: MỤC E - KIẾN NGHỊ & ĐỀ XUẤT (TIÊU ĐỀ MỤC)
  // ==========================================
  const h4 = form.addSectionHeaderItem();
  h4.setTitle('📝 MỤC E: KIẾN NGHỊ & ĐỀ XUẤT')
    .setHelpText('Ý kiến đóng góp ý kiến về môi trường làm việc, trao đổi học thuật, quy trình công việc');

  form.addParagraphTextItem().setTitle('E. Ý kiến đóng góp & Kiến nghị khác').setHelpText('Ví dụ:\n(1) Cần trao đổi học tập 1 tuần / 1 lần các kiến thức mới.\n(2) Hồ sơ cá nhân thiết kế phải in giấy và trình chủ nhiệm xem trước khi xuất.').setRequired(false);

  form.setConfirmationMessage('Cảm ơn bạn đã gửi thông tin cập nhật hồ sơ năng lực cá nhân. Dữ liệu đã được ghi nhận vào hệ thống quản lý.');

  Logger.log('=== RESULT URLS ===');
  Logger.log('PUBLISHED_URL: ' + form.getPublishedUrl());
  Logger.log('EDIT_URL: ' + form.getEditUrl());
}
